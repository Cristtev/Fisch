-- CHARACTER HELPERS
local function GetCharacter()
    return LocalPlayer.Character
end

local function GetHumanoid()
    local Character = GetCharacter()

    if not Character then
        return nil
    end

    return Character:FindFirstChildOfClass("Humanoid")
end

local function GetRoot()
    local Character = GetCharacter()

    if not Character then
        return nil
    end

    return Character:FindFirstChild("HumanoidRootPart")
end

-- PLAYER MOVEMENT
local function SetWalkSpeed(Value)
    Value = math.clamp(
        math.round(tonumber(Value) or 16),
        16,
        500
    )
    PlayerState.WalkSpeed = Value
    local Humanoid = GetHumanoid()
    if Humanoid then
        Humanoid.WalkSpeed = Value
    end
end

local function SetJumpPower(Value)
    Value = math.clamp(
        math.round(tonumber(Value) or 16),
        16,
        500
    )
    
    PlayerState.JumpPower = Value

    local Humanoid = GetHumanoid()

    if Humanoid then
        Humanoid.UseJumpPower = true
        Humanoid.JumpPower = Value
    end
end

    Humanoid.UseJumpPower = true

    Humanoid.JumpPower =

-- FLY


local FlyConnection

local function StopFly()

    if FlyConnection then
        FlyConnection:Disconnect()
        FlyConnection = nil
    end

    local Humanoid = GetHumanoid()

    if Humanoid then
        Humanoid.AutoRotate = true
    end
end

local function StartFly()
    StopFly()
    FlyConnection =
    RunService.RenderStepped:Connect(function()
        if not PlayerState.Fly then
            return
        end

    local Root = GetRoot()
    local Humanoid = GetHumanoid()
    if not Root or not Humanoid then
        return
    end

    local Camera =
    workspace.CurrentCamera
    if not Camera then
        return
    end

    Humanoid.AutoRotate = false

    local Direction =
    Humanoid.MoveDirection
    local Vertical = 0
    if UserInputService:IsKeyDown(
        Enum.KeyCode.Space
    )
    then
        Vertical = 1
    elseif UserInputService:IsKeyDown(
        Enum.KeyCode.LeftControl
    )
    then
        Vertical = -1
    end
    
    Direction =
    Direction +
    Vector3.new(0, Vertical, 0)
    if Direction.Magnitude > 0 then
        Root.AssemblyLinearVelocity =
        Direction.Unit * 70
        else
            Root.AssemblyLinearVelocity =
            Vector3.zero
            end
        end)
    end

local function SetFly(State)

    PlayerState.Fly = State

    if State then
        StartFly()
    else
        StopFly()
        local Root = GetRoot()
        if Root then
            Root.AssemblyLinearVelocity =
            Vector3.zero
        end
    end
end


-- NOCLIP


local NoclipConnection

local function StartNoclip()

    if NoclipConnection then
        NoclipConnection:Disconnect()
    end

    NoclipConnection =
        RunService.Stepped:Connect(function()
            if not PlayerState.Noclip then
                return
        end
        
        local Character =
        GetCharacter()
        if not Character then
            return
        end

            for _, Object in ipairs(
                Character:GetDescendants()
            ) do

                if Object:IsA("BasePart") then
                    Object.CanCollide = false
                end
            end
        end)
end

local function StopNoclip()

    if NoclipConnection then
        NoclipConnection:Disconnect()
        NoclipConnection = nil
    end

    local Character =
        GetCharacter()

    if not Character then
        return
    end

    for _, Object in ipairs(
        Character:GetDescendants()
    ) do

        if Object:IsA("BasePart") then
            Object.CanCollide = true
        end
    end
end

local function SetNoclip(State)

    PlayerState.Noclip = State

    if State then
        StartNoclip()
    else
        StopNoclip()
    end
end


-- LIGHTWEIGHT PLAYER ESP


local ESPFolder =
    Instance.new("Folder")

ESPFolder.Name =
    "LocalPlayerESP"

ESPFolder.Parent =
    workspace

local ESPObjects = {}

local function RemoveESP(Player)

    local Highlight =
        ESPObjects[Player]

    if Highlight then
        Highlight:Destroy()
        ESPObjects[Player] = nil
    end
end

local function AddESP(Player)

    if Player == LocalPlayer then
        return
    end

    if ESPObjects[Player] then
        return
    end

    local Character =
        Player.Character

    if not Character then
        return
    end

    local Highlight =
        Instance.new("Highlight")

    Highlight.Name =
        "ESP_" .. Player.UserId

    Highlight.Adornee =
        Character

    Highlight.DepthMode =
        Enum.HighlightDepthMode.AlwaysOnTop

    Highlight.FillTransparency =
        0.75

    Highlight.OutlineTransparency =
        0

    Highlight.Parent =
        ESPFolder

    ESPObjects[Player] =
        Highlight
end

local function RefreshESP()

    if not PlayerState.ESP then

        for Player in pairs(ESPObjects) do
            RemoveESP(Player)
        end

        return
    end

    for _, Player in ipairs(
        Players:GetPlayers()
    ) do

        AddESP(Player)
    end
end

local function SetESP(State)

    PlayerState.ESP = State

    RefreshESP()
end

Players.PlayerAdded:Connect(function(Player)

    Player.CharacterAdded:Connect(function()

        if PlayerState.ESP then
            task.wait(0.2)
            AddESP(Player)
        end
    end)

    if PlayerState.ESP then
        AddESP(Player)
    end
end)

Players.PlayerRemoving:Connect(function(Player)

    RemoveESP(Player)

    if PlayerState.SelectedPlayer ==
        Player then

        PlayerState.SelectedPlayer = nil
    end
end)


-- TP TO PLAYER


local function TeleportToPlayer(Player)

    if not Player then
        return
    end

    local TargetCharacter =
        Player.Character

    local TargetRoot =
        TargetCharacter
        and TargetCharacter:FindFirstChild(
            "HumanoidRootPart"
        )

    local Root =
        GetRoot()

    if not Root or not TargetRoot then
        return
    end

    Root.CFrame =
        TargetRoot.CFrame *
        CFrame.new(0, 0, 4)
end

-- REEL SYSTEM


local function StartReel()

    if ReelState.Active then
        return
    end

    ReelState.Active = true
    ReelState.Progress = 0
    ReelState.Success = false

    if ReelState.Mode == "Normal" then

        print("Normal reel started")

    elseif ReelState.Mode == "Legit" then

        print("Legit reel started")

    elseif ReelState.Mode == "Instant" then

        print("Instant reel started")

        ReelState.Progress = 1
        ReelState.Success = true
        ReelState.Active = false
    end
end

local function FinishReel(Success)

    ReelState.Active = false
    ReelState.Progress =
        Success and 1 or 0

    ReelState.Success =
        Success
end

local function UpdateReel(DeltaTime)

    if not ReelState.Active then
        return
    end

    if ReelState.Mode == "Normal" then
        if HoldingReel then
            ReelState.Progress = ReelState.Progress + (DeltaTime * 0.35)
            else
                ReelState.Progress = ReelState.Progress - (DeltaTime * 0.12)
                end

    elseif ReelState.Mode == "Legit" then

        if HoldingReel then

            ReelState.Progress = ReelState.Progress +
                (DeltaTime * 0.42)

        else

            ReelState.Progress = ReelState -
                (DeltaTime * 0.07)
        end
    end

    ReelState.Progress =
        math.clamp(
            ReelState.Progress,
            0,
            1
        )

    if ReelState.Progress >= 1 then
        FinishReel(true)

    elseif ReelState.Progress <= 0 then
        FinishReel(false)
    end
end

RunService.RenderStepped:Connect(UpdateReel)

UserInputService.InputBegan:Connect(
    function(Input, Processed)

        if Processed then
            return
        end

        if Input.KeyCode ==
            Enum.KeyCode.Space then

            HoldingReel = true

            if not ReelState.Active then
                StartReel()
            end
        end
    end
)

UserInputService.InputEnded:Connect(
    function(Input)

        if Input.KeyCode ==
            Enum.KeyCode.Space then

            HoldingReel = false
        end
    end
)

            Button.Text =

        end

                Callback(Option)
            end
        )
    end

    task.defer(RefreshCanvas)

    return Frame
end

-- player list
local function BuildPlayerList()

    local Options = {}

    for _, Player in ipairs(
        Players:GetPlayers()
    ) do

        if Player ~= LocalPlayer then
            table.insert(
                Options,
                Player.Name
            )
        end
    end

    table.sort(Options)

    return Options
end

local PlayerSelector

local function RebuildPlayerSelector()

    if PlayerSelector then
        PlayerSelector:Destroy()
    end

    PlayerSelector =
        CreateScrollSelector(
            PlayerCategory,
            "TP TO PLAYER",
            BuildPlayerList(),
            function(Name)

                local Target =
                    Players:FindFirstChild(
                        Name
                    )

                if Target then

                    PlayerState.SelectedPlayer =
                        Target

                    TeleportToPlayer(
                        Target
                    )
                end
            end,
            125
        )
end

RebuildPlayerSelector()

Players.PlayerAdded:Connect(function()
    task.wait()
    RebuildPlayerSelector()
end)

Players.PlayerRemoving:Connect(function()
    task.wait()
    RebuildPlayerSelector()
end)
