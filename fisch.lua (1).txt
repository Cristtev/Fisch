local function ShowCategory(name)
    for categoryName, frame in pairs(Categories) do
        frame.Visible = (categoryName == name)
    end
    for buttonName, button in pairs(CategoryButtons) do
        if buttonName == name then
            button.BackgroundColor3 = COLORS.CYAN
            button.TextColor3 = COLORS.DARK
        else
            button.BackgroundColor3 = COLORS.PANEL
            button.TextColor3 = COLORS.TEXT
        end
    end
end

local function CreateSlider(parent, name, min, default, max, callback)
    local Container = Instance.new("Frame")
    Container.Size = UDim2.new(1, 0, 0, 55)
    Container.BackgroundColor3 = COLORS.PANEL
    Container.BorderSizePixel = 0
    Container.Parent = parent

    local Corner = Instance.new("UICorner")
    Corner.CornerRadius = UDim.new(0, 6)
    Corner.Parent = Container

    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(1, -20, 0, 20)
    Label.Position = UDim2.new(0, 10, 0, 6)
    Label.BackgroundTransparency = 1
    Label.Font = Enum.Font.GothamSemibold
    Label.Text = name .. ": " .. default
    Label.TextColor3 = COLORS.TEXT
    Label.TextSize = 11
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.Parent = Container

    local SliderBg = Instance.new("Frame")
    SliderBg.Size = UDim2.new(1, -20, 0, 6)
    SliderBg.Position = UDim2.new(0, 10, 0, 34)
    SliderBg.BackgroundColor3 = COLORS.DARK
    SliderBg.BorderSizePixel = 0
    SliderBg.Parent = Container

    local BgCorner = Instance.new("UICorner")
    BgCorner.CornerRadius = UDim.new(1, 0)
    BgCorner.Parent = SliderBg

    local percentage = (default - min) / (max - min)
    local SliderFill = Instance.new("Frame")
    SliderFill.Size = UDim2.new(percentage, 0, 1, 0)
    SliderFill.BackgroundColor3 = COLORS.CYAN
    SliderFill.BorderSizePixel = 0
    SliderFill.Parent = SliderBg

    local FillCorner = Instance.new("UICorner")
    FillCorner.CornerRadius = UDim.new(1, 0)
    FillCorner.Parent = SliderFill

    local dragging = false
    local function SetValueFromX(x)
        local alpha = math.clamp((x - SliderBg.AbsolutePosition.X) / SliderBg.AbsoluteSize.X, 0, 1)
        SliderFill.Size = UDim2.new(alpha, 0, 1, 0)
        local value = math.floor(min + ((max - min) * alpha))
        Label.Text = name .. ": " .. value
        callback(value)
    end

    SliderBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            SetValueFromX(input.Position.X)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            SetValueFromX(input.Position.X)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local function StartReel()
    if ReelState.Active then return end
    ReelState.Active = true
    ReelState.Progress = 0
    ReelState.Success = false

    if ReelState.Mode == "Instant" then
        ReelState.Progress = 1
        ReelState.Success = true
        ReelState.Active = false
    end
end

local function FinishReel(success)
    ReelState.Active = false
    ReelState.Progress = success and 1 or 0
    ReelState.Success = success
end

local function UpdateReel(deltaTime)
    if not ReelState.Active then return end

    if ReelState.Mode == "Normal" then
        if ReelState.Holding then
            ReelState.Progress = ReelState.Progress + (deltaTime * 0.35)
        else
            ReelState.Progress = ReelState.Progress - (deltaTime * 0.12)
        end
    elseif ReelState.Mode == "Legit" then
        if ReelState.Holding then
            ReelState.Progress = ReelState.Progress + (deltaTime * 0.42)
        else
            ReelState.Progress = ReelState.Progress - (deltaTime * 0.07)
        end
    end

    ReelState.Progress = math.clamp(ReelState.Progress, 0, 1)

    if ReelState.Progress >= 1 then
        FinishReel(true)
    elseif ReelState.Progress <= 0 and not ReelState.Holding then
        FinishReel(false)
    end
end

RunService.RenderStepped:Connect(function(deltaTime)
    UpdateReel(deltaTime)
    ProgressBarFill.Size = UDim2.new(ReelState.Progress, 0, 1, 0)

    if ReelState.Active then
        StatusLabel.Text = string.format("REELING... (%d%%) | MODE: %s", math.floor(ReelState.Progress * 100), ReelState.Mode:upper())
        StatusLabel.TextColor3 = COLORS.CYAN
    elseif ReelState.Success then
        StatusLabel.Text = "REEL COMPLETE // SUCCESS"
        StatusLabel.TextColor3 = COLORS.SUCCESS
    else
        StatusLabel.Text = string.format("STATUS: READY | MODE: %s", ReelState.Mode:upper())
        StatusLabel.TextColor3 = COLORS.MUTED
    end
end)

local function SafeTeleport(targetCFrame)
    local character = player.Character
    if not character then return end
    local root = character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    -- Safe arrival calculation: Offset 50 studs back along LookVector to avoid NPC overlap
    local safeOffset = targetCFrame.LookVector * -50 + Vector3.new(0, 3, 0)
    local destination = targetCFrame + safeOffset

    root.CFrame = destination
end

    end)
end

    if not PlayerState.InfiniteJump then return end
    local character = player.Character
    if character and character:FindFirstChildOfClass("Humanoid") then
        character:FindFirstChildOfClass("Humanoid"):ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)

RunService.RenderStepped:Connect(function()
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end

    if humanoid.WalkSpeed ~= PlayerState.WalkSpeed then
        humanoid.WalkSpeed = PlayerState.WalkSpeed
    end

    humanoid.UseJumpPower = true
    if humanoid.JumpPower ~= PlayerState.JumpPower then
        humanoid.JumpPower = PlayerState.JumpPower
    end

    if PlayerState.HopJump and humanoid.FloorMaterial ~= Enum.Material.Air then
        humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    end

    if PlayerState.Noclip then
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end
end)

-- Dragging System (Desktop & Touch)
local dragging = false
local dragStart = nil
local startPosition = nil

TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = MainFrame.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

    if mobileDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - mobileDragStart
        if delta.Magnitude > 8 then mobileMoved = true end
        MobileButton.Position = UDim2.new(
            mobileStartPos.X.Scale,
            mobileStartPos.X.Offset + delta.X,
            mobileStartPos.Y.Scale,
            mobileStartPos.Y.Offset + delta.Y
        )
    end
end)

    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        if mobileDragging and not mobileMoved then
            ToggleUI()
        end
        mobileDragging = false
    end
end)
